<script type="module">
import { CodeHub } from "https://app.codehub.site.je/sdk.js";

const APP_TOKEN = "TESTE";

const $ = id => document.getElementById(id);

const sdk = new CodeHub({ appToken: APP_TOKEN });

function status(msg, cls = "", spinner = false) {
  const s = $("status");
  s.className = cls;
  s.innerHTML = spinner ? `<span class="spinner"></span> ${msg}` : msg;
}

function setAvatar(photoUrl, name) {
  const el = $("avatar");
  const initial = (name || "?").trim().charAt(0).toUpperCase() || "?";
  if (photoUrl) {
    el.innerHTML = `<img src="${photoUrl}" alt="avatar" onerror="this.style.display='none';this.parentElement.innerHTML='<span class="fallback">${initial}</span>'">`;
  } else {
    el.innerHTML = `<span class="fallback">${initial}</span>`;
  }
}

function showAccount(conta, authId) {
  $("name").textContent = conta.name || "(sem nome)";
  $("email").textContent = conta.email || "(sem e-mail)";
  $("uid").textContent = conta.uid || "—";
  $("verified").textContent = conta.emailVerified ? "Sim" : "Não";
  $("authid").textContent = authId || "—";
  $("loc").textContent = [conta.city, conta.state].filter(Boolean).join(" / ") || "—";
  setAvatar(conta.photoUrl, conta.name);
  $("profile").classList.add("show");
}

function hideAccount() {
  $("profile").classList.remove("show");
  $("token").textContent = "Nenhum status carregado.";
}

async function recarregarConta(silencioso = false) {
  if (!silencioso) status("Carregando sessão...", "pend", true);
  try {
    const authId = sdk.readSession();
    if (!authId) {
      status("Nenhuma sessão salva. Faça login primeiro.");
      hideAccount();
      return null;
    }
    const st = await sdk.getAuthStatus(authId);
    $("token").textContent = JSON.stringify(st, null, 2);
    if (!st || st.status !== "confirmed") {
      status("Sessão não confirmada: " + ((st && st.status) || "desconhecido"), "err");
      return null;
    }
    const conta = await sdk.getAccount(authId);
    showAccount(conta, authId);
    if (!silencioso) status("Conta carregada com sucesso.", "ok");
    return { ok: true, status: st.status, conta, authId };
  } catch (e) {
    if (!silencioso) status("Erro: " + (e.message || e), "err");
    console.error(e);
    return null;
  }
}

sdk.on("pending", () => status("Aguardando autenticação...", "pend", true));
sdk.on("confirmed", async () => {
  status("Login confirmado! Carregando perfil...", "ok");
  await recarregarConta(true);
});
sdk.on("error", e => status("Erro: " + (e.message || e), "err"));

$("btnLogin").addEventListener("click", async () => {
  try { status("Abrindo login...", "pend", true); await sdk.login(undefined, true); }
  catch (e) { status("Erro: " + (e.message || e), "err"); }
});

$("btnAuthByApp").addEventListener("click", async () => {
  try {
    status("Autenticando via App Token...", "pend", true);
    const d = await sdk.authByApp();
    if (!d.ok) throw new Error(d.erro || "Falha");
    status("AuthID criado via App Token!", "ok");
    await recarregarConta(true);
  } catch (e) { status("Erro: " + (e.message || e), "err"); console.error(e); }
});

$("btnRefresh").addEventListener("click", () => recarregarConta(false));

$("btnLogout").addEventListener("click", async () => {
  try {
    const r = await sdk.logout();
    if (r.ok) { hideAccount(); status("Sessão removida.", "ok"); }
    else { status("Cancelado ou falhou: " + r.msg, "err"); }
  } catch (e) { status("Erro: " + (e.message || e), "err"); }
});

$("btnCopy").addEventListener("click", async () => {
  const valor = $("token").textContent;
  try { await navigator.clipboard.writeText(valor); status("Status copiado.", "ok"); }
  catch { status("Não foi possível copiar.", "err"); }
});

(async () => {
  console.log("SDK v" + sdk.version + " carregado com appToken:", APP_TOKEN);
  if (!sdk.readSession()) {
    status("Nenhuma sessão. Faça login ou use App Token.");
    hideAccount();
    return;
  }
  await recarregarConta(false);
})();
</script>

Métodos do SDK v1.5

  • sdk.login() — abre popup e aguarda login
  • sdk.authByApp() — cria AuthID confirmado via appToken
  • sdk.getAuthStatus(authId) — lê o status da sessão
  • sdk.getProfile(authId) — pega foto, nome, e-mail
  • sdk.getAccount(authId) — perfil formatado (uid, name, photoUrl, etc)
  • sdk.reloadAccount() — recarrega o perfil inteiro
  • sdk.logout() — apaga o authId do banco e da memória
  • sdk.version — retorna "1.5"
⚠️ Sobre o iframe: o site de teste precisa permitir ser embutido (não pode enviar X-Frame-Options: DENY nem frame-ancestors 'none'). Se não abrir, é porque o servidor está bloqueando.
